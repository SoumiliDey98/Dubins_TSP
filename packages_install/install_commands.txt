# ==============================================================================
# Setup & Installation Commands for Multi-Agent Dubins TSP Simulation
# ==============================================================================

# Step 1: Update system packages (Linux / Ubuntu / Debian)
sudo apt update

# Step 2: Install Python 3, pip, and venv support (if not already installed)
sudo apt install -y python3 python3-pip python3-venv

# Step 3: Create a virtual environment
python3 -m venv venv

# Step 4: Activate the virtual environment
source venv/bin/activate

# Step 5: Upgrade pip
pip install --upgrade pip

# Step 6: Install required Python dependencies
pip install numpy scipy matplotlib networkx shapely pillow

# (Or alternatively using requirements.txt):
# pip install -r requirements.txt

# Step 7: Run the simulation
python3 main.py
